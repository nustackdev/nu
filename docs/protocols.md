
# Interfaces

## TKV

```python
# Iterator Protocol (low-level, like LMDB cursor / RocksDB iterator)
class IteratorProtocol(Protocol):
    def seek(self, key: TupleKey) -> bool: ...
    def seek_to_first(self) -> bool: ...
    def seek_to_last(self) -> bool: ...
    def next(self) -> bool: ...
    def prev(self) -> bool: ...
    def key(self) -> TupleKey: ...
    def value(self) -> Value: ...
    def is_valid(self) -> bool: ...
    def close(self) -> None: ...
    def __enter__(self) -> IteratorProtocol: ...
    def __exit__(self, exc_type, exc_val, exc_tb) -> None: ...

# Storage Protocol
class StorageProtocol(Protocol):
    def begin(self, *, write: bool = False) -> TransactionProtocol: ...
    def close(self) -> None: ...

# Transaction Protocol
class TransactionProtocol(Protocol):
    # Point access
    def get(self, key: TupleKey, default: Value | None = None) -> Value: ...
    def put(self, key: TupleKey, value: Value) -> None: ...
    def delete(self, key: TupleKey) -> bool: ...
    def has(self, key: TupleKey) -> bool: ...

    # Scan access
    def iterator(self) -> IteratorProtocol: ...
    def range_delete(self, start: TupleKey, end: TupleKey, *, start_inclusive: bool = True, end_inclusive: bool = False) -> int: ...

    # Batch access
    def multiget(self, keys: list[TupleKey]) -> dict[TupleKey, Value]: ...

    # Transaction lifecycle
    def commit(self) -> None: ...
    def abort(self) -> None: ...

    @property
    def writable(self) -> bool: ...

    def __enter__(self) -> TransactionProtocol: ...
    def __exit__(self, exc_type, exc_val, exc_tb) -> None: ...
```

## Tree

```python
# Tree Protocol
class TreeProtocol(Protocol):
    # Container operations
    def make_container(self, path: TupleKey, *, structure_id: int | None = None) -> None: ...
    def is_container(self, path: TupleKey) -> bool: ...
    def has_children(self, path: TupleKey) -> bool: ...
    def delete_subtree(self, path: TupleKey) -> int: ...

    # Navigation
    def children(self, path: TupleKey) -> list[TupleKey]: ...
    def descendants(self, path: TupleKey) -> list[TupleKey]: ...
    def parent(self, path: TupleKey) -> TupleKey | None: ...
    def child_keys(self, path: TupleKey) -> list[str | int | bytes]: ...

    # Value operations
    def get(self, path: TupleKey, default: Value | None = None) -> Value: ...
    def put(self, path: TupleKey, value: Value) -> None: ...
    def delete(self, path: TupleKey) -> bool: ...
    def has(self, path: TupleKey) -> bool: ...

    # Metadata
    def get_structure_id(self, path: TupleKey) -> int | None: ...
    def set_structure_id(self, path: TupleKey, structure_id: int) -> None: ...

    # Counts
    def count_children(self, path: TupleKey) -> int: ...
    def count_descendants(self, path: TupleKey) -> int: ...
```
